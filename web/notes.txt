export function useCurrentUser() {
  const { token } = useAuthState();

  const { data: user, ...rest } = useQuery({
    queryKey: ["me", token],
    queryFn: token ? apiRoutes.getMe : skipToken,
    staleTime: Infinity
  });

  return { user, ...rest };
}
